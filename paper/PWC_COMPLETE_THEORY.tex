\section{Appendix: the zero-parameter rotation code}
\label{app:code}
The full pipeline is one command in the repository
(\texttt{python PWC/run\_full\_framework.py}). The rotation core, with both
constants derived (no fitted parameters):

\begin{lstlisting}[language=Python, caption={PWC rotation core --- derived $a_0$ and $s$, no fitted parameters}]
import numpy as np

# locked constants
c0   = 2.99792458e8          # m/s, the medium's speed
rho0 = 8.74e-27              # kg/m^3, the medium's rest density
Ybow = 5.93e26               # Pa, the yield (0.5 rho_max (954 km/s)^2)
s    = 1.0/19.0              # the stretch (retained-heat fraction)
melt = 0.05                  # the 5% still tied (the fuel gauge)

# derived constants (no fitted parameters)
a0 = 6.61e-11                # m/s^2 = a_hold(rho0), the medium's holding
s_shared = 1.5 * 3.0 * melt  # 0.2250 = the mist's ram (3/2 x 3 x 0.05)
g = (1.0/melt)**(1.0/16.0)   # 1.2059 = the coupling, 16-stage loop

def pwc_rotation(g_bar, shared, locked):
    """g_obs = g_bar + sqrt(a0 g_bar) [1 + s_shared * shared * (1 - locked)]"""
    return g_bar + np.sqrt(a0*g_bar) * (1.0 + s_shared*shared*(1.0-locked))

def C_of_rho(rho):
    """the universal speed limit: C(rho) = min(c0, sqrt(2 Ybow / rho))"""
    return min(c0, np.sqrt(2.0*Ybow/rho))

# example: a PROBES-scale galaxy, g_bar = 1e-10 m/s^2, shared = 1, locked = 0
g_pred = pwc_rotation(1e-10, 1.0, 0.0)
print(f"derived coupling g     = {g:.6f}   (16 pi^3 alpha/3 = 1.206740)")
print(f"derived s_shared      = {s_shared:.4f}   (fitted 0.2264)")
print(f"example g_obs         = {g_pred:.3e} m/s^2")
print(f"RBH-1 cap check       = {C_of_rho(1.304e15)/1e3:.0f} km/s (954)")
\end{lstlisting}
